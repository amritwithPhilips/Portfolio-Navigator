function table(rows,scope,editable){
 const groups=new Map();for(const r of rows){const g=r.org||'BURF not supplied';if(!groups.has(g))groups.set(g,[]);groups.get(g).push(r);}
 const summary=(rs,g,id,footer=false)=>{const a=AOPViews.aggregate(rs),open=expanded.has(id),bandClass=a.band==='Healthy'?'healthy':a.band==='Contain'?'contain':'mixed';
  const amount=(field,label)=>'<td class="numeric" data-aggregate="'+field+'" data-exact="'+(a[field].sum??'')+'"><strong>'+dec(a[field].sum)+'</strong><small>Avg '+dec(a[field].mean)+'</small></td>';
  return '<tr class="'+(footer?'summary-total':'group burf-summary')+'" data-group-summary="'+esc(id)+'" data-organisation="'+esc(g)+'"><td class="identity">'+(footer?'<b>'+esc(g)+'</b>':'<button class="burf-toggle" data-collapse="'+esc(id)+'" aria-expanded="'+open+'"><span class="triangle">'+(open?'▼':'▶')+'</span>'+esc(g)+'</button>')+' <span class="group-count">('+a.funded+'/'+a.count+')</span></td><td><span class="band-label '+bandClass+'">'+esc(a.band)+'</span><small>'+Object.entries(a.bands).map(([k,v])=>esc(k)+' '+v).join(' · ')+'</small></td><td class="numeric" data-aggregate="score" data-exact="'+(a.score.mean??'')+'"><span class="score-badge '+(a.score.mean>=session.settings.merit?'strong-score':'low-score')+'">'+dec(a.score.mean)+'</span><small>Average merit</small></td>'+amount('cost')+amount('final')+amount('unfunded')+amount('benefit')+'<td class="numeric" data-aggregate="evidence"><strong>'+a.evidence+'</strong><small>of '+a.count+' numeric claims</small></td><td><b>'+a.funded+' funded · '+(a.count-a.funded)+' unfunded</b><small>'+a.partial+' partial scope · '+a.held+' evidence/eligibility holds</small><small>'+a.missingCost+' missing requests</small></td></tr>';
 };
 const groupRows=[...groups].sort((a,b)=>(AOPViews.aggregate(b[1]).score.mean??-1)-(AOPViews.aggregate(a[1]).score.mean??-1)||a[0].localeCompare(b[0]));
 return '<div class="table-actions"><p>BURF totals remain visible when closed. Score is the unweighted average merit; money is summed, with the average below. Missing amounts are excluded from their average. Evid. counts numeric submitted benefit claims, not validated benefits. ('+rows.length+' displayed programs)</p><button data-expand-all="'+scope+'">Expand all</button><button data-close-all="'+scope+'">Collapse all</button></div><div class="tablebox burf-tablebox" data-table-scope="'+scope+'"><table class="portfolio burf-table"><thead><tr><th class="identity">BU | Region | Function / Program</th><th>Decision Band</th><th>Score</th><th>Requested (M€)</th><th>Prioritised (M€)</th><th>Unfunded (M€)</th><th>Benefits (M€)</th><th>Evid.</th><th>'+(scope==='unfunded'?'Reason not funded / demand details':'Funding status / demand details')+'</th></tr></thead><tbody>'+groupRows.map(([g,rs])=>{
  const id=scope+'|'+encodeURIComponent(g),hidden=expanded.has(id)?'':' hidden';
  return summary(rs,g,id)+(editable?'<tr class="group-controls" data-group-row="'+esc(id)+'"'+hidden+'><td colspan="9"><label><input type="checkbox" aria-label="Select displayed '+esc(g)+'" data-group-check="'+esc(g)+'"> Select displayed programs</label><button data-group="'+esc(g)+'" data-choice="include">Include '+rs.filter(r=>r.cost>0).length+' eligible amount rows</button><button data-group="'+esc(g)+'" data-choice="exclude">Exclude displayed group</button><small>Changes are manual conference decisions. Missing/nonpositive requests cannot be included.</small></td></tr>':'')+rs.map(r=>'<tr class="program-row" data-group-row="'+esc(id)+'" data-row-id="'+esc(r.id)+'"'+hidden+'><td class="identity"><div class="program-identity">'+(editable?'<input type="checkbox" aria-label="Include '+esc(r.id)+'" data-include="'+esc(r.id)+'" '+(r.final>0?'checked':'')+' '+(!(r.cost>0)?'disabled':'')+'>':'')+'<button class="program" data-program="'+esc(r.id)+'">'+esc(r.name||'Program name not supplied')+'</button></div><small>'+esc(r.id)+' · '+esc(r.sourceSheet)+' row '+r.sourceRow+'</small></td><td><span class="band-label '+(AOP.key(r.decisionBand)==='healthy'?'healthy':AOP.key(r.decisionBand)==='contain'?'contain':'mixed')+'">'+esc(r.decisionBand||'Unassessed')+'</span></td><td class="numeric"><span class="score-badge '+(r.score>=session.settings.merit?'strong-score':'low-score')+'">'+dec(r.score)+'</span></td><td class="numeric">'+dec(r.cost)+'</td><td class="numeric">'+(editable?'<input class="amount" aria-label="Prioritised amount '+esc(r.id)+'" type="number" step="0.1" min="0" data-amount="'+esc(r.id)+'" data-original="'+r.final+'" value="'+dec(r.final)+'" '+(!(r.cost>0)?'disabled':'')+'>':dec(r.final))+'</td><td class="numeric">'+dec(r.unfunded)+'</td><td class="numeric">'+dec(r.benefit)+'<small>'+(r.ratio===null?'Multiple unavailable':dec(r.ratio)+'x of request')+'</small></td><td class="numeric">'+(r.benefit!==null?'1':'0')+'</td><td><b class="status-pill">'+esc(r.status)+'</b><small>'+esc(r.reason)+'</small><small>'+esc(r.newCarry==='Change & Sustain'?'Sustain Engineering':r.newCarry||'Not supplied')+' · '+esc(r.stageName)+'</small><small>'+esc(r.overall||'Overall decision unassessed')+'</small><small>'+esc(r.benefitDetails||AOP.clean(r.benefitRaw)||'Benefit evidence missing')+'</small><small class="warning">'+esc(r.warnings.join('; '))+'</small>'+(editable?'<button data-auto="'+esc(r.id)+'">Return to automatic</button><details class="inline-review"><summary>Note / evidence review</summary><label>Optional decision note<input data-note="'+esc(r.id)+'" value="'+esc(r.override?.note||'')+'"></label><label><input type="checkbox" data-review-benefit="'+esc(r.id)+'" '+(r.review?.benefitReviewed?'checked':'')+'> Above-5.0x claim reviewed</label><label>Contain condition<input data-condition="'+esc(r.id)+'" value="'+esc(r.review?.condition||'')+'"></label><label>Captured Consequence tier<select data-consequence="'+esc(r.id)+'"><option value="">Automated interpretation</option>'+[0,5,10,15].map(v=>'<option value="'+v+'" '+(r.review?.consequence===v?'selected':'')+'>'+dec(v)+'</option>').join('')+'</select></label><label>Review rationale<input data-review-note="'+esc(r.id)+'" value="'+esc(r.review?.note||'')+'"></label><button data-apply-review="'+esc(r.id)+'">Record evidence review</button></details>'+(r.override?.state==='needs-review'?'<button data-resolve="'+esc(r.id)+'">Reconfirm previous decision</button><button data-discard="'+esc(r.id)+'">Discard previous decision</button>':''):'')+'</td></tr>').join('');
 }).join('')+(rows.length?'':'<tr><td colspan="9" class="empty">No matching programs. Clear filters to return to the conference view.</td></tr>')+'</tbody><tfoot>'+summary(rows,'All displayed BURFs',scope+'|total',true)+'</tfoot></table></div>';
}
